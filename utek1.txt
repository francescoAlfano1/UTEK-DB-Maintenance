ALTER PROCEDURE [dbo].[RPISP_UtekCleanActLog] 

AS
BEGIN
  -- Inizio della transazione  
    BEGIN TRANSACTION;  

    BEGIN TRY  
        -- Cancellazione delle righe provenienti dal Modula e tracciate sul Log
        DELETE FROM dbo.actlog
        WHERE SUBSTRING(LTRIM(RTRIM(al_key)), 21, 1) = 'M';  -- ' identifico le righe provenienti dal Modula'
        
        -- Invocazione del comando SHRINK  
      --  DBCC SHRINKFILE ('dbo.actlog', 2000);     nome e poi dimensione 
         -- DBCC SHRINKDATABASE ('nome_del_databaseLOG.db');  
        
        -- Completamento della transazione  
        COMMIT TRANSACTION;  
    END TRY  
    BEGIN CATCH  
        -- Se trova errori rollback  
        ROLLBACK TRANSACTION;  
        -- Stampa l'errore  
        DECLARE @ErrorMessage NVARCHAR(4000), @ErrorSeverity INT, @ErrorState INT;  
        SELECT @ErrorMessage = ERROR_MESSAGE(),  
               @ErrorSeverity = ERROR_SEVERITY(),  
               @ErrorState = ERROR_STATE();  
        RAISERROR (@ErrorMessage, @ErrorSeverity, @ErrorState);  
    END CATCH  
END;
